import { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  return {
 rules: [
  {
    userAgent: "*",
    allow: "/",
    disallow: [
      "/admin",
      "/auth",
      "/en/cards",
      "/jp/cards",
      "/en/packs",
      "/jp/packs",
    ],
  },
],
    sitemap: "https://www.tcgfinderjapan.com/sitemap.xml",
  };
}
